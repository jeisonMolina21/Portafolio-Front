// Inicialización de API REST & Servicio de Carnetización
import express, { Request, Response } from 'express';
import { Pool } from 'pg';

const app = express();
const db = new Pool({ connectionString: process.env.DATABASE_URL });

app.get('/api/v1/passes/:id', async (req: Request, res: Response) => {
  const { id } = req.params;
  const result = await db.query('SELECT * FROM passes WHERE user_id = $1', [id]);
  return res.json({ status: 'active', data: result.rows[0] });
});
